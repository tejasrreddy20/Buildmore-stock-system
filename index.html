  <script>

    const STORAGE_KEY = "buildmore_inventory_v1";

    let currentFilter = "all";


    // Load initial stock from localStorage (or seed sample items on first run)

    let stockItems = JSON.parse(localStorage.getItem(STORAGE_KEY)) || [

      { id: "1", sku: "TIL-101", name: "Glossy Vitrified Tile 2x4", category: "Tiles", qty: 45, minQty: 15 },

      { id: "2", sku: "SAN-202", name: "Ceramic Wall Hung Closet", category: "Sanitaryware", qty: 6, minQty: 10 },

      { id: "3", sku: "PLM-305", name: "CPVC Pipe 1 Inch (10ft)", category: "Plumbing", qty: 80, minQty: 20 }

    ];


    function persistAndRender() {

      localStorage.setItem(STORAGE_KEY, JSON.stringify(stockItems));

      render();

    }


    function setFilter(filter, btn) {

      currentFilter = filter;

      document.querySelectorAll(".nav-btn").forEach(b => b.classList.remove("active"));

      if (btn) btn.classList.add("active");

      document.getElementById("viewTitle").textContent =

        filter === "low" ? "Active Low Stock Alerts" : "Stock Inventory";

      render();

    }


    function render() {

      const query = document.getElementById("searchInput").value.trim().toLowerCase();

      const tbody = document.getElementById("stockTableBody");


      const lowStockCount = stockItems.filter(i => Number(i.qty) <= Number(i.minQty)).length;

      const totalUnits = stockItems.reduce((sum, i) => sum + Number(i.qty), 0);


      document.getElementById("statTotal").textContent = stockItems.length;

      document.getElementById("statUnits").textContent = totalUnits;

      document.getElementById("statLow").textContent = lowStockCount;

      document.getElementById("alertPill").textContent = `${lowStockCount} Active Stock Alerts`;


      const filtered = stockItems.filter(item => {

        const matchesSearch =

          item.sku.toLowerCase().includes(query) ||

          item.name.toLowerCase().includes(query) ||

          item.category.toLowerCase().includes(query);

        const matchesFilter =

          currentFilter === "all" || Number(item.qty) <= Number(item.minQty);

        return matchesSearch && matchesFilter;

      });


      if (filtered.length === 0) {

        tbody.innerHTML = `<tr><td colspan="7" style="text-align:center; color:#64748b; padding:24px;">No stock items found.</td></tr>`;

        return;

      }


      tbody.innerHTML = filtered.map(item => {

        const isLow = Number(item.qty) <= Number(item.minQty);

        return `

          <tr>

            <td><strong>${escapeHtml(item.sku)}</strong></td>

            <td>${escapeHtml(item.name)}</td>

            <td>${escapeHtml(item.category)}</td>

            <td><strong>${item.qty}</strong></td>

            <td>${item.minQty}</td>

            <td>

              <span class="badge ${isLow ? "badge-low" : "badge-ok"}">

                ${isLow ? "Low Stock" : "In Stock"}

              </span>

            </td>

            <td>

              <button class="action-btn" onclick="adjustQty('${item.id}', 1)">+1</button>

              <button class="action-btn" onclick="adjustQty('${item.id}', -1)">-1</button>

              <button class="action-btn" onclick="editItem('${item.id}')">Edit</button>

              <button class="action-btn delete" onclick="deleteItem('${item.id}')">Delete</button>

            </td>

          </tr>

        `;

      }).join("");

    }


    // Requirement 4: Immediate Save & Update

    function saveItem(event) {

      event.preventDefault();

      const id = document.getElementById("itemId").value;

      const newItem = {

        id: id || String(Date.now()),

        sku: document.getElementById("sku").value.trim(),

        name: document.getElementById("name").value.trim(),

        category: document.getElementById("category").value,

        qty: Math.max(0, Number(document.getElementById("qty").value)),

        minQty: Math.max(0, Number(document.getElementById("minQty").value))

      };


      if (id) {

        stockItems = stockItems.map(item => (item.id === id ? newItem : item));

      } else {

        stockItems = [newItem, ...stockItems];

      }


      persistAndRender();

      closeModal();

    }


    function adjustQty(id, delta) {

      stockItems = stockItems.map(item =>

        item.id === id ? { ...item, qty: Math.max(0, Number(item.qty) + delta) } : item

      );

      persistAndRender();

    }


    // Requirement 4: Immediate Delete

    function deleteItem(id) {

      stockItems = stockItems.filter(item => item.id !== id);

      persistAndRender();

    }


    function editItem(id) {

      const item = stockItems.find(i => i.id === id);

      if (!item) return;

      document.getElementById("modalTitle").textContent = "Edit Stock Item";

      document.getElementById("itemId").value = item.id;

      document.getElementById("sku").value = item.sku;

      document.getElementById("name").value = item.name;

      document.getElementById("category").value = item.category;

      document.getElementById("qty").value = item.qty;

      document.getElementById("minQty").value = item.minQty;

      document.getElementById("itemModal").classList.add("open");

    }


    function openModal() {

      document.getElementById("modalTitle").textContent = "Add Stock Item";

      document.getElementById("stockForm").reset();

      document.getElementById("itemId").value = "";

      document.getElementById("itemModal").classList.add("open");

    }


    function closeModal() {

      document.getElementById("itemModal").classList.remove("open");

    }


    function escapeHtml(str) {

      return String(str)

        .replace(/&/g, "&amp;")

        .replace(/</g, "&lt;")

        .replace(/>/g, "&gt;")

        .replace(/"/g, "&quot;");

    }


    render();
